actions quietly Link bind NEEDLIBS LINKFLAGS
{
  call $(LINK) $(LINKFLAGS) $(1)
  call_1core $(LINK) $(1)
  call_1core_filtered $(LINK) $(1)
}

actions Clean
{
  :label-1
  if errorlevel 1 goto label-1
  rem nothing $(1) here
  REM an upper-case remark
  del $(1)
  echo #( plain text )# then #/(out\$(1))/#
}
